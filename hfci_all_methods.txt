# ----------------------------------------------------------------------------- 1. HELPER FUNCTIONS (survey-weighted)
def wmean(X, w):                                     # survey-weighted column means
    return (X * w[:, None]).sum(0) / w.sum()         # sum(weight x value) / sum(weights)

def wstd(X, w):                                      # survey-weighted standard deviations
    m = wmean(X, w)                                  # weighted means
    return np.sqrt((((X - m) ** 2) * w[:, None]).sum(0) / (w.sum() - 1))   # weighted variance, then square root

def wcorr(Z, w):                                     # survey-weighted correlation matrix of standardised data
    C = (Z * w[:, None]).T @ Z / w.sum()             # weighted covariance of z-scores
    d = np.sqrt(np.diag(C))                          # weighted standard deviations (close to 1)
    return C / np.outer(d, d)                        # scale to correlations

def varimax(L, it=200, tol=1e-9):                    # varimax rotation of a loading matrix L (p indicators x k factors)
    p, k = L.shape; R = np.eye(k); d = 0             # p = indicators, k = factors, R = rotation matrix, d = previous criterion
    for _ in range(it):                              # iterate until the criterion stops improving
        Lr = L @ R                                   # current rotated loadings
        u, s, vt = np.linalg.svd(L.T @ (Lr**3 - Lr @ np.diag((Lr**2).sum(0)) / p))   # SVD of the varimax gradient
        R = u @ vt; dn = s.sum()                     # update the rotation; dn = new value of the criterion
        if dn < d * (1 + tol): break                 # stop when the improvement is negligible
        d = dn                                       # remember the criterion for the next pass
    return L @ R                                     # return the rotated loadings

def fix_sign(L):                                     # make each factor's loadings mostly positive (sign is arbitrary)
    return L * np.where(L.sum(0) < 0, -1, 1)         # flip any column whose loadings sum to a negative number

def assign_weights(L, cols):                         # shared by FA-only and Hybrid: loadings -> weights
    assign = np.abs(L).argmax(1)                     # assign each indicator to the factor with its largest |loading|
    ss = (L ** 2).sum(0); fw = ss / ss.sum()         # sum of squared loadings per factor; fw = factor weights
    wi = np.zeros(L.shape[0])                        # container for indicator weights
    for f in range(L.shape[1]):                      # loop over factors
        m = assign == f                              # indicators assigned to factor f
        if m.any(): wi[m] = (L[m, f] ** 2) / (L[m, f] ** 2).sum() * fw[f]   # within-factor share x factor weight
    return pd.Series(wi / wi.sum(), index=cols)      # normalise so weights sum to 1

def paf(R, k, it=500, tol=1e-6):                     # principal-axis factoring (common-factor model)
    h = 1 - 1 / np.diag(np.linalg.inv(R))            # starting communalities: squared multiple correlations
    for _ in range(it):                              # iterate communalities until they converge
        Rr = R.copy(); np.fill_diagonal(Rr, h)       # reduced correlation matrix: communalities on the diagonal
        w, v = np.linalg.eigh(Rr); o = np.argsort(w)[::-1][:k]            # keep the k largest eigenvalues
        L = v[:, o] * np.sqrt(np.clip(w[o], 0, None))                     # factor loadings (negative eigenvalues set to 0)
        hn = np.clip((L ** 2).sum(1), 0, 0.995)      # new communalities, capped to avoid Heywood cases
        if np.max(np.abs(hn - h)) < tol: h = hn; break                    # converged: stop
        h = hn                                       # otherwise carry the new communalities forward
    return L, h                                      # loadings and communalities

def parallel_threshold(p, n_eff, reps=N_PA, seed=3):  # Horn's parallel analysis: eigenvalues expected from pure noise
    rng = np.random.default_rng(seed)                # reproducible random generator
    sims = np.array([np.sort(np.linalg.eigvalsh(np.corrcoef(rng.normal(size=(int(n_eff), p)).T)))[::-1] for _ in range(reps)])  # random-data eigenvalues
    return np.percentile(sims, 95, axis=0)           # 95th percentile threshold for each component

def eig_sorted(R):                                   # eigen-decomposition of a correlation matrix, largest first
    w, v = np.linalg.eigh(R); o = np.argsort(w)[::-1]
    return w[o], v[:, o]

def weights_from_corr(R, method, k=None, cols=None):  # weights for any method from a (weighted) correlation matrix
    p = R.shape[0]; cols = cols if cols is not None else list(range(p))   # number of indicators and labels
    w_, v_ = eig_sorted(R)                           # eigenvalues and eigenvectors
    k = k or int((w_ > 1).sum())                     # number of factors: Kaiser rule unless supplied
    if method == 'equal':                            # METHOD 1: equal weights
        return pd.Series(np.ones(p) / p, index=cols)
    if method == 'pca':                              # METHOD 2: PCA only
        L = fix_sign(v_[:, :k] * np.sqrt(w_[:k]))    # unrotated loadings = eigenvector x sqrt(eigenvalue)
        share = w_[:k] / w_[:k].sum()                # each component's share of the retained variance
        raw = (L ** 2 * share).sum(1)                # squared loadings weighted by variance share, summed over components
        return pd.Series(raw / raw.sum(), index=cols)    # normalise to sum 1
    if method == 'fa':                               # METHOD 3: factor analysis only
        L, h = paf(R, k)                             # principal-axis factoring
        L = fix_sign(varimax(L)) if k > 1 else fix_sign(L)   # rotate only if more than one factor
        return assign_weights(L, cols)               # loadings -> weights
    if method == 'hybrid':                           # METHOD 4: hybrid PCA-FA
        L = fix_sign(varimax(v_[:, :k] * np.sqrt(w_[:k])))   # PCA loadings, then varimax rotation
        return assign_weights(L, cols)               # within- and between-factor weights

def simulate_households(seed=SEED, N=N_HH, R_=N_ROUNDS):   # builds household microdata from five hidden household factors
    rng = np.random.default_rng(seed)                # reproducible randomness
    Cf = np.array([[1,.4,.3,.35,.4],[.4,1,.2,.3,.35],[.3,.2,1,.3,.3],[.35,.3,.3,1,.4],[.4,.35,.3,.4,1]])   # correlation between hidden factors
    Lc = np.linalg.cholesky(Cf)                      # used to create correlated factors
    Lam = {'H1':[.8,.1,0,0,.1],'H2':[.75,.15,0,0,.15],'H3':[.7,.1,.05,.1,.1],'H4':[.6,0,0,.25,.2],     # true loadings of each indicator on the hidden factors
           'H5':[.2,.75,0,.1,.1],'H6':[.15,.7,.05,.05,.2],'H7':[.2,.7,.05,.1,.15],
           'H8':[0,.05,.8,.1,.1],'H9':[0,.1,.85,.05,.1],'H10':[.05,.1,.7,.1,.2],'H11':[.1,.15,.6,.1,.25],
           'H12':[.15,.1,.05,.8,.1],'H13':[.1,.05,.05,.8,.1],'H14':[.1,.1,.1,.7,.2],
           'H15':[.25,.2,.2,.15,.65],'H16':[.1,.05,.05,.2,.6],'H17':[.05,.05,.25,.1,.65],'H18':[.2,.15,.1,.15,.55]}
    shift = np.array([0.0, -0.25, -0.45, -0.15])     # change in average conditions by round: deterioration, then partial recovery
    frames = []                                      # one DataFrame per round
    for r in range(R_):                              # loop over survey rounds
        Fh = rng.normal(size=(N, 5)) @ Lc.T + shift[r]   # household factors for this round
        d = {}                                       # indicator columns
        for c in COLS:                               # build each indicator
            better = Fh @ np.array(Lam[c]) + rng.normal(0, rng.uniform(.5, .75), N)   # indicator in the "better" direction plus noise
            d[c] = POL[c] * better                   # record as a survey would (negative indicators: higher = worse)
        df = pd.DataFrame(d); df['round'] = r + 1    # add the round number
        df['w'] = np.exp(rng.normal(0, .5, N)); df['w'] *= N / df['w'].sum()   # survey weights scaled to sum to N per round
        frames.append(df)
    out = pd.concat(frames, ignore_index=True)       # stack the four rounds
    out.insert(0, 'hh_id', range(1, len(out) + 1))   # household identifier
    return out

def cronbach(cs):                                    # (i) reliability of a pillar (survey-weighted)
    idx = [COLS.index(c) for c in cs]; k = len(cs)   # positions of the pillar's indicators
    S = Z[:, idx]; var_sum = (wstd(S, w) ** 2).sum() # sum of the indicator variances
    tot_var = wstd(S.sum(1, keepdims=True), w)[0] ** 2   # variance of the pillar total
    return k / (k - 1) * (1 - var_sum / tot_var)     # Cronbach's alpha formula
